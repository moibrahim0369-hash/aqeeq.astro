/**
 * محرك توليد تقارير مصفوفة القدر - عقيق أسترو
 * الربط الآلي بين المعادلات الرياضية وملفات النصوص التفسيرية
 */

import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';
import { calculateDestinyMatrix } from './حسابات-مصفوفة-القدر.js';

// ضبط مسارات المجلدات
const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

// مسار مجلد النصوص بالنسبة لمحركات الحسابات
const BASE_DATA_PATH = path.join(__dirname, '..', 'قاعدة-البيانات-والنصوص');
const ENERGIES_PATH = path.join(BASE_DATA_PATH, 'طاقات-مصفوفة-القدر');
const TAILS_PATH = path.join(BASE_DATA_PATH, 'الذيول-الكارمية-المركبة');

/**
 * دالة مساعدة لقراءة ملف Markdown بناءً على رقم الطاقة
 */
function readEnergyFile(energyNumber) {
  try {
    const files = fs.readdirSync(ENERGIES_PATH);
    // البحث عن الملف الذي يبدأ برقم الطاقة (مثلاً: 07- أو 7-)
    const formattedNum = energyNumber.toString().padStart(2, '0');
    const matchedFile = files.find(file => 
      file.startsWith(formattedNum) || file.startsWith(`${energyNumber}-`)
    );

    if (!matchedFile) return `تفسير الطاقة ${energyNumber} غير متوفر حالياً.`;
    return fs.readFileSync(path.join(ENERGIES_PATH, matchedFile), 'utf-8');
  } catch (error) {
    return `خطأ في تحميل ملف الطاقة ${energyNumber}: ${error.message}`;
  }
}

/**
 * دالة مساعدة لقراءة ملف الذيل الكارمي المركب
 */
function readKarmicTailFile(tailCode) {
  try {
    const files = fs.readdirSync(TAILS_PATH);
    // البحث عن الملف الذي يحتوي على كود الذيل (مثلاً: 18-6-5)
    const matchedFile = files.find(file => file.includes(tailCode));

    if (!matchedFile) return `تفسير الذيل الكارمي (${tailCode}) غير متوفر حالياً.`;
    return fs.readFileSync(path.join(TAILS_PATH, matchedFile), 'utf-8');
  } catch (error) {
    return `خطأ في تحميل ملف الذيل الكارمي ${tailCode}: ${error.message}`;
  }
}

/**
 * الدالة التنفيذية الكبرى لتوليد التقرير الشامل للعميل
 */
export function generateClientDestinyReport(day, month, year, clientName = "العميل") {
  // 1. تشغيل المحرك الحسابي
  const matrixData = calculateDestinyMatrix(day, month, year);

  // 2. جلب النصوص المطابقة لكل موضع مصيري
  const personalityReport = readEnergyFile(matrixData.corePoints.left);
  const spiritReport = readEnergyFile(matrixData.corePoints.top);
  const coreComfortReport = readEnergyFile(matrixData.corePoints.center);
  const karmicTailReport = readKarmicTailFile(matrixData.karmicTail.code);
  const loveChannelReport = readEnergyFile(matrixData.channels.love);
  const moneyChannelReport = readEnergyFile(matrixData.channels.money);

  // 3. بناء هيكل التقرير النهائي المنظم
  return {
    meta: {
      clientName,
      birthDate: matrixData.birthDate,
      calculatedAt: new Date().toISOString(),
      platform: "عقيق أسترو - Akeek Astro"
    },
    matrixNumbers: matrixData,
    interpretations: {
      personality: {
        title: "طاقة الشخصية والهوية الخارجية",
        energyNumber: matrixData.corePoints.left,
        content: personalityReport
      },
      spirituality: {
        title: "بوابة الروح والاتصال الإلهي",
        energyNumber: matrixData.corePoints.top,
        content: spiritReport
      },
      coreComfort: {
        title: "جوهر الروح ونقطة الراحة النفسية",
        energyNumber: matrixData.corePoints.center,
        content: coreComfortReport
      },
      karmicTail: {
        title: "عقدة الذيل الكارمي والديون الماضية",
        tailCode: matrixData.karmicTail.code,
        content: karmicTailReport
      },
      channels: {
        love: {
          title: "بوابة الحب والشريك المثالي",
          energyNumber: matrixData.channels.love,
          content: loveChannelReport
        },
        money: {
          title: "بوابة الوفرة والتدفق المالي",
          energyNumber: matrixData.channels.money,
          content: moneyChannelReport
        }
      }
    }
  };
}
