إليك كود مكون رسم خريطة مصفوفة القدر الهندسية (Destiny Matrix SVG).
الكود مكتوب بصيغة SVG متجاوب وفائق الدقة، يرسم:
 * المربع المستقيم (سماء وأرض / الأنا).
 * المربع المائل بزاوية 45° (مربع الأجداد والسلالة).
 * خطوط الطاقة الداخلية وأقطار التوازن.
 * دوائر الطاقات الملونة بالألوان الأصلية المعتمدة لنظام الأركانا ومصفوفة القدر (الأزرق، البنفسجي، الأصفر، الأحمر، الأخضر).
 * توزيع الأرقام بدقة بالغة في إحداثياتها الهندسية بناءً على مخرجات دالة الحسابات.
اسم الملف: رسم-مصفوفة-القدر.js
(ضعه داخل مجلد: محركات-الحسابات/ أو مجلد مكونات الواجهة)
/**
 * محرك رسم مصفوفة القدر التفاعلية (SVG Visualizer) - منصة عقيق أسترو
 * يُنتج كود SVG نقي وعالي الدقة لعرضه مباشرة في المتصفح أو حفظه كصورة
 */

export function renderDestinyMatrixSVG(matrixData) {
  const { corePoints, karmicTail, channels, ancestralSquare } = matrixData;

  // إعدادات لوحة الرسم
  const size = 600;
  const center = size / 2;
  const radius = 220; // نصف قطر الدائرة المحيطية
  const ancestralRadius = radius * 0.707; // نصف قطر مربع الأجداد

  // حساب الإحداثيات الهندسية للنقاط الأساسية
  const coords = {
    top: { x: center, y: center - radius }, // الأعلى (الروح / 20 سنة)
    bottom: { x: center, y: center + radius }, // الأسفل (الكارما / 60 سنة)
    left: { x: center - radius, y: center }, // اليسار (الشخصية / 0 سنة)
    right: { x: center + radius, y: center }, // اليمين (المادة / 40 سنة)
    center: { x: center, y: center }, // المركز (جوهر الروح)

    // نقاط مربع الأجداد المائل
    topLeft: { x: center - ancestralRadius, y: center - ancestralRadius },
    topRight: { x: center + ancestralRadius, y: center - ancestralRadius },
    bottomLeft: { x: center - ancestralRadius, y: center + ancestralRadius },
    bottomRight: { x: center + ancestralRadius, y: center + ancestralRadius },

    // نقاط الذيل الكارمي السفلي
    karmicMid: { x: center, y: center + radius * 0.65 },
    karmicSub: { x: center, y: center + radius * 0.82 },

    // خط العلاقات والمال
    love: { x: center + radius * 0.25, y: center + radius * 0.35 },
    money: { x: center + radius * 0.45, y: center + radius * 0.2 },
    balance: { x: center + radius * 0.35, y: center + radius * 0.28 }
  };

  return `
  <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${size} ${size}" width="100%" height="100%" style="background: #0f111a; font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;">
    <defs>
      <!-- تدرجات لونية للفخامة العقيقية -->
      <radialGradient id="matrixGlow" cx="50%" cy="50%" r="50%">
        <stop offset="0%" stop-color="#4a154b" stop-opacity="0.3"/>
        <stop offset="100%" stop-color="#0f111a" stop-opacity="0"/>
      </radialGradient>
      
      <!-- ستايل النصوص والأرقام -->
      <style>
        .matrix-line { stroke: #334155; stroke-width: 1.5; stroke-dasharray: 4 4; }
        .solid-line { stroke: #64748b; stroke-width: 2; }
        .channel-money { stroke: #10b981; stroke-width: 2.5; }
        .channel-love { stroke: #ec4899; stroke-width: 2.5; }
        .node-text { font-size: 15px; font-weight: bold; fill: #ffffff; text-anchor: middle; dominant-baseline: central; }
        .center-text { font-size: 18px; font-weight: 900; fill: #fbbf24; text-anchor: middle; dominant-baseline: central; }
      </style>
    </defs>

    <!-- توهج الخلفية -->
    <circle cx="${center}" cy="${center}" r="${radius + 40}" fill="url(#matrixGlow)"/>

    <!-- 1. المحاور والأقطار الرئيسية -->
    <line x1="${coords.left.x}" y1="${coords.left.y}" x2="${coords.right.x}" y2="${coords.right.y}" class="solid-line"/>
    <line x1="${coords.top.x}" y1="${coords.top.y}" x2="${coords.bottom.x}" y2="${coords.bottom.y}" class="solid-line"/>
    
    <!-- خطوط مربع الأجداد القطرية -->
    <line x1="${coords.topLeft.x}" y1="${coords.topLeft.y}" x2="${coords.bottomRight.x}" y2="${coords.bottomRight.y}" class="matrix-line"/>
    <line x1="${coords.topRight.x}" y1="${coords.topRight.y}" x2="${coords.bottomLeft.x}" y2="${coords.bottomLeft.y}" class="matrix-line"/>

    <!-- 2. المربعان الهندسيان (النجمة الثمانية) -->
    <!-- المربع المستقيم (سماء وأرض) -->
    <polygon points="
      ${coords.top.x},${coords.top.y} 
      ${coords.right.x},${coords.right.y} 
      ${coords.bottom.x},${coords.bottom.y} 
      ${coords.left.x},${coords.left.y}" 
      fill="none" stroke="#e2e8f0" stroke-width="2"/>

    <!-- المربع المائل (الأجداد والسلالة) -->
    <polygon points="
      ${coords.topLeft.x},${coords.topLeft.y} 
      ${coords.topRight.x},${coords.topRight.y} 
      ${coords.bottomRight.x},${coords.bottomRight.y} 
      ${coords.bottomLeft.x},${coords.bottomLeft.y}" 
      fill="none" stroke="#ef4444" stroke-width="1.8" stroke-opacity="0.8"/>

    <!-- 3. مسار خط الحب والمال -->
    <line x1="${coords.karmicMid.x}" y1="${coords.karmicMid.y}" x2="${coords.right.x - 50}" y2="${coords.right.y + 50}" class="solid-line" stroke="#94a3b8"/>

    <!-- 4. الدوائر العقدية وتوزيع الطاقات (الألوان الرسمية) -->
    
    <!-- الأعلى: الروح (بنفسجي/أزرق ناعم) -->
    <circle cx="${coords.top.x}" cy="${coords.top.y}" r="18" fill="#6366f1"/>
    <text x="${coords.top.x}" y="${coords.top.y}" class="node-text">${corePoints.top}</text>

    <!-- اليسار: الشخصية والميلاد (بنفسجي ساطع) -->
    <circle cx="${coords.left.x}" cy="${coords.left.y}" r="18" fill="#8b5cf6"/>
    <text x="${coords.left.x}" y="${coords.left.y}" class="node-text">${corePoints.left}</text>

    <!-- اليمين: المادة والسنة (أحمر قرمزي) -->
    <circle cx="${coords.right.x}" cy="${coords.right.y}" r="18" fill="#f43f5e"/>
    <text x="${coords.right.x}" y="${coords.right.y}" class="node-text">${corePoints.right}</text>

    <!-- الأسفل: قاعدة الكارما (أحمر مظلم) -->
    <circle cx="${coords.bottom.x}" cy="${coords.bottom.y}" r="18" fill="#991b1b"/>
    <text x="${coords.bottom.x}" y="${coords.bottom.y}" class="node-text">${corePoints.bottom}</text>

    <!-- عقد الذيل الكارمي السفلي -->
    <circle cx="${coords.karmicMid.x}" cy="${coords.karmicMid.y}" r="14" fill="#b91c1c"/>
    <text x="${coords.karmicMid.x}" y="${coords.karmicMid.y}" class="node-text" style="font-size:12px;">${karmicTail.numbers[1]}</text>

    <circle cx="${coords.karmicSub.x}" cy="${coords.karmicSub.y}" r="14" fill="#dc2626"/>
    <text x="${coords.karmicSub.x}" y="${coords.karmicSub.y}" class="node-text" style="font-size:12px;">${karmicTail.numbers[2]}</text>

    <!-- أركان الأجداد الأربعة -->
    <circle cx="${coords.topLeft.x}" cy="${coords.topLeft.y}" r="15" fill="#3b82f6"/>
    <text x="${coords.topLeft.x}" y="${coords.topLeft.y}" class="node-text" style="font-size:13px;">${ancestralSquare.fatherSpiritual}</text>

    <circle cx="${coords.topRight.x}" cy="${coords.topRight.y}" r="15" fill="#ec4899"/>
    <text x="${coords.topRight.x}" y="${coords.topRight.y}" class="node-text" style="font-size:13px;">${ancestralSquare.motherSpiritual}</text>

    <circle cx="${coords.bottomLeft.x}" cy="${coords.bottomLeft.y}" r="15" fill="#3b82f6"/>
    <text x="${coords.bottomLeft.x}" y="${coords.bottomLeft.y}" class="node-text" style="font-size:13px;">${ancestralSquare.fatherMaterial}</text>

    <circle cx="${coords.bottomRight.x}" cy="${coords.bottomRight.y}" r="15" fill="#ec4899"/>
    <text x="${coords.bottomRight.x}" y="${coords.bottomRight.y}" class="node-text" style="font-size:13px;">${ancestralSquare.motherMaterial}</text>

    <!-- عقد خطي الحب والمال -->
    <!-- نقطة الحب (وردي) -->
    <circle cx="${coords.love.x}" cy="${coords.love.y}" r="13" fill="#ec4899"/>
    <text x="${coords.love.x}" y="${coords.love.y}" class="node-text" style="font-size:11px;">${channels.love}</text>

    <!-- نقطة المال (أخضر زمردي) -->
    <circle cx="${coords.money.x}" cy="${coords.money.y}" r="13" fill="#10b981"/>
    <text x="${coords.money.x}" y="${coords.money.y}" class="node-text" style="font-size:11px;">${channels.money}</text>

    <!-- نقطة الاتزان المشترك (برتقالي ذهبي) -->
    <circle cx="${coords.balance.x}" cy="${coords.balance.y}" r="13" fill="#f59e0b"/>
    <text x="${coords.balance.x}" y="${coords.balance.y}" class="node-text" style="font-size:11px;">${channels.balance}</text>

    <!-- 5. نقطة المركز (جوهر الروح / شمس صفراء مضيئة) -->
    <circle cx="${coords.center.x}" cy="${coords.center.y}" r="22" fill="#fbbf24" stroke="#d97706" stroke-width="3"/>
    <text x="${coords.center.x}" y="${coords.center.y}" class="center-text">${corePoints.center}</text>

    <!-- توقيع المنصة في زاوية الرسم -->
    <text x="${size - 20}" y="${size - 20}" font-size="11" fill="#475569" text-anchor="end">عقيق أسترو | Akeek Astro</text>
  </svg>
  `;
}

كيفية استخدامه وعرضه في الواجهة (HTML / React / Frontend):
 * في صفحة ويب عادية (HTML/JS):
import { calculateDestinyMatrix } from './محركات-الحسابات/حسابات-مصفوفة-القدر.js';
import { renderDestinyMatrixSVG } from './محركات-الحسابات/رسم-مصفوفة-القدر.js';

// 1. حساب الأرقام
const data = calculateDestinyMatrix(25, 9, 1999);

// 2. توليد وعرض الرسم في أي div بالصفحة
const svgString = renderDestinyMatrixSVG(data);
document.getElementById('matrix-container').innerHTML = svgString;

 * في تطبيق React / Next.js:
<div 
  className="w-full max-w-xl mx-auto shadow-2xl rounded-2xl overflow-hidden"
  dangerouslySetInnerHTML={{ __html: renderDestinyMatrixSVG(data) }} 
/>

بهذا يكتمل الجانب البصري، ويظهر الرسم الهندسي بدقة متناهية وألوان متباينة مع أرقام تاريخ الميلاد في مواضعها الصحيحة.
